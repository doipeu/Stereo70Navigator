                <p>
                    Aceasta este o mica aplicatie cu ajutorul careia se poate determina pozitia unui imobil
                    (cu precizie metrica), prin introducerea a maxim 4 puncte de coordonate.
                    Tot ce trebuie sa faceti este sa completati in formularul din dreapta
                    valorile pentru coordonatele <i>X(Nord)</i> si <i>Y(Est)</i> aferente fiecarui punct,
                    iar dupa ce ati verificat datele apasati butonul
                    <i class="text-theme-color font-size14 text-uppercase">Localizare</i>.
                </p>
                <p>
                    Trebuie mentionat ca aceste coordonate sunt exprimate in metri si au ca sistem de referinta Sistemul de Proiectie Stereografic 1970 (Stereo70).
                </p>
                <p>
                    Exemplu de coordonate pentru un punct: X(Nord) = <i>329976.803</i> ; Y(Est) = <i>589696.632</i>
                </p>
                <p>
                    <b>Observatie</b>: pentru a pozitiona cu precizie aceste puncte de coordonate,
                    aveti nevoie de serviciile unui inginer geodez/topograf autorizat.

            // Add current coordinates to array
            coords.push(geoCode);

            // Extend bounds
            bounds.extend(geoCode);

            // Create Google Maps Marker
            let Gmarker = new google.maps.Marker({
                map: Gmap,
                position: geoCode,
                icon: '/assets/img/map/marker_small.png',
                draggable: false,
                title: 'Punctul ' + point.name
            });

        </a>
    </p>
</div>
`;

            let Ginfowindow = new google.maps.InfoWindow({
                content: contentString
            });

            google.maps.event.addListener(Gmarker, 'click', function() {
                Ginfowindow.open(Gmap,Gmarker);
            });
        });

    // Add new bounds object to map
    if (coordinates.length) {
        Gmap.fitBounds(bounds);
    }

    // Draw line between 2 points
    if (coords.length === 2) {
        let line = new google.maps.Polyline({
            path: coords,
            geodesic: true,
            strokeColor: '#FF0000',
            strokeOpacity: 1.0,
            strokeWeight: 2
        });

        line.setMap(Gmap);
    }

    // Draw polygon
    if (coords.length > 2) {
        const polygon = new google.maps.Polygon({
